\section{Memory Bank}

\begin{itemize}
\item $\epsilon = -N \Delta \phi_m /\Delta t$ ... Faraday's Law
\item $\epsilon = -L \Delta I /\Delta t$ ... Faraday's Law
\item $\phi_m = \vec{B} \cdot \vec{A} = BA \cos(\theta)$ ... Definition of magnetic flux
\item $\vec{B} = \mu_0 n I \hat{k}$ ... Magnetic field in a solenoid of length $L$, turns $N$, current $I$, and $n = N/L$.
\item 1 H = 1 $\Omega$ s ... The \textit{Henry} is a unit of inductance, often represented by $L$.
\item $\mu_0 = 4\pi \times 10^{-7}$ ... T m$^{-1}$ A$^{-1}$.
\item $I(t) = I_0\left(1 - \exp(-t/\tau)\right)$ ... Current when RL circuits turn on ($\tau = L/R$).
\item $I(t) = I_0\exp(-t/\tau)$ ... Current when RL circuits turn off ($\tau = L/R$).
\end{itemize}
