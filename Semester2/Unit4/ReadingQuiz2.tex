\section{Memory Bank}

\begin{itemize}
\item $\epsilon = -N \Delta \phi_m /\Delta t$ ... Faraday's Law
\item $\phi_m = \vec{B} \cdot \vec{A}$ ... Definition of magnetic flux
\item 1 T = 1 N A$^{-1}$ m$^{-1}$ ... Definition of 1 Tesla
\item $\epsilon = N A B \omega$ ... Peak emf of electric generator
\end{itemize}

\section{Faraday's Law}

\begin{enumerate}
\item Suppose a uniform B-field of 0.1 T passes through a wire loop with an area of 0.05 m$^2$.  Calculate $\phi_m$, the magnetic flux, if the angle between the B-field and loop area is (a) 0 degrees, (b) 45 degrees, and (c) 90 degrees.  (d) Suppose the B-field drops to 0 T in 1 ms.  What is the induced emf? \\ \vspace{2cm}
\item Verify that the units of $\Delta \phi_m/\Delta t$ are volts. \\ \vspace{2cm}
\item A coil is moved through a magnetic field as shown in Fig. \ref{fig:1}. The field is uniform inside the rectangle and zero outside. What is the direction of the induced current and what is the direction of the magnetic force on the coil at each position shown? \\ \vspace{2cm}
\end{enumerate}

\section{Electric Generators}

\begin{enumerate}
\item (a) A bicycle generator rotates at 1875 rad/s, producing an 18.0 V peak emf. It has a 1.00 by 3.00 cm rectangular coil in a 0.640 T field. How many turns are in the coil? (b) Is this number of turns of wire practical for a 1.00 by 3.00 cm coil? \\ \vspace{4cm}
\item Design a generator that uses just 25 turns of wire in the coil, but still produces 18 V. \\ \vspace{2cm}
\item The magnetic field through a circular loop of radius 10.0 cm varies with time as shown in Fig. \ref{fig:2}.  The field is perpendicular to the loop.  Plot the magnitude of the induced emf in the loop as a function of time.
\end{enumerate}
